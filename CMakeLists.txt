cmake_minimum_required(VERSION 3.22)
project(VocalBalance VERSION 0.1.1 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(VB_BUILD_PLUGIN "Build JUCE VST3 and standalone" ON)
option(VB_BUILD_VST2_COMPAT "Build independent 64-bit VST2 compatibility module" ON)
enable_testing()
add_executable(dsp_tests tests/DspTests.cpp)
target_include_directories(dsp_tests PRIVATE Source)
add_test(NAME dsp COMMAND dsp_tests)
if(VB_BUILD_PLUGIN)
  if(DEFINED JUCE_SOURCE_DIR)
    add_subdirectory(${JUCE_SOURCE_DIR} JUCE)
  else()
    include(FetchContent)
    FetchContent_Declare(JUCE GIT_REPOSITORY https://github.com/juce-framework/JUCE.git GIT_TAG 29396c22c93392d6738e021b83196283d6e4d850 GIT_SHALLOW FALSE)
    FetchContent_MakeAvailable(JUCE)
  endif()
  set(VB_FORMATS VST3 Standalone)
  set(VB_VST2_SDK_PATH "" CACHE PATH "Legally obtained VST2 SDK root (optional)")
  if(VB_VST2_SDK_PATH)
    if(NOT EXISTS "${VB_VST2_SDK_PATH}/pluginterfaces/vst2.x/aeffect.h")
      message(FATAL_ERROR "VST2 SDK must contain pluginterfaces/vst2.x/aeffect.h")
    endif()
    juce_set_vst2_sdk_path(${VB_VST2_SDK_PATH})
    list(APPEND VB_FORMATS VST)
  endif()
  if(CMAKE_CROSSCOMPILING)
    set(VB_AUTO_MANIFEST FALSE)
  else()
    set(VB_AUTO_MANIFEST TRUE)
  endif()
  juce_add_plugin(VocalBalance VST3_AUTO_MANIFEST ${VB_AUTO_MANIFEST} COMPANY_NAME "Independent Audio" BUNDLE_ID org.independentaudio.vocalbalance PLUGIN_MANUFACTURER_CODE InAu PLUGIN_CODE Vb01 FORMATS ${VB_FORMATS} PRODUCT_NAME "Vocal Balance" IS_SYNTH FALSE NEEDS_MIDI_INPUT FALSE NEEDS_MIDI_OUTPUT FALSE COPY_PLUGIN_AFTER_BUILD FALSE)
  target_sources(VocalBalance PRIVATE Source/Plugin.cpp Source/Editor.cpp)
  target_compile_definitions(VocalBalance PUBLIC JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_VST3_CAN_REPLACE_VST2=0)
  if(VB_BUILD_VST2_COMPAT AND NOT VB_VST2_SDK_PATH)
    get_target_property(VB_MODULES juce::juce_audio_processors INTERFACE_JUCE_MODULE_PATH)
    add_library(VocalBalance_VST2 MODULE Source/Vst2Adapter.cpp)
    target_include_directories(VocalBalance_VST2 PRIVATE Source third_party "${VB_MODULES}")
    target_compile_definitions(VocalBalance_VST2 PRIVATE
      $<TARGET_PROPERTY:VocalBalance,COMPILE_DEFINITIONS>)
    target_link_libraries(VocalBalance_VST2 PRIVATE VocalBalance)
    if(WIN32 AND TARGET VocalBalance_rc_lib)
      target_link_libraries(VocalBalance_VST2 PRIVATE VocalBalance_rc_lib)
    endif()
    set_target_properties(VocalBalance_VST2 PROPERTIES
      PREFIX "" OUTPUT_NAME "Vocal Balance"
      LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/VocalBalance_artefacts/$<CONFIG>/VST2"
      RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/VocalBalance_artefacts/$<CONFIG>/VST2")
  endif()
  juce_add_console_app(plugin_tests PRODUCT_NAME "Vocal Balance Tests")
  target_sources(plugin_tests PRIVATE tests/PluginTests.cpp)
  target_include_directories(plugin_tests PRIVATE Source)
  target_compile_definitions(plugin_tests PRIVATE JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_VST3_CAN_REPLACE_VST2=0)
  target_link_libraries(plugin_tests PRIVATE VocalBalance juce::juce_audio_utils)
  add_test(NAME plugin COMMAND plugin_tests)
  if(WIN32 OR CMAKE_SYSTEM_NAME STREQUAL "Linux")
    get_target_property(VB_MODULES juce::juce_audio_processors INTERFACE_JUCE_MODULE_PATH)
    if(EXISTS "${VB_MODULES}/juce_audio_processors_headless/format_types/VST3_SDK")
      set(VB_SDK "${VB_MODULES}/juce_audio_processors_headless/format_types/VST3_SDK")
    else()
      set(VB_SDK "${VB_MODULES}/juce_audio_processors/format_types/VST3_SDK")
    endif()
    add_executable(vst3_smoke tests/Vst3Smoke.cpp
      "${VB_SDK}/pluginterfaces/base/coreiids.cpp"
      "${VB_SDK}/pluginterfaces/base/funknown.cpp"
      "${VB_SDK}/public.sdk/source/vst/vstinitiids.cpp")
    target_include_directories(vst3_smoke PRIVATE "${VB_SDK}")
    target_link_libraries(vst3_smoke PRIVATE ${CMAKE_DL_LIBS})
    add_dependencies(vst3_smoke VocalBalance_VST3)
    add_test(NAME vst3_module COMMAND vst3_smoke "$<TARGET_FILE:VocalBalance_VST3>")
    if(TARGET VocalBalance_VST2)
      if(NOT WIN32)
        find_package(X11 REQUIRED)
      endif()
      add_executable(vst2_smoke tests/Vst2Smoke.cpp)
      target_include_directories(vst2_smoke PRIVATE third_party ${X11_INCLUDE_DIR})
      target_link_libraries(vst2_smoke PRIVATE ${CMAKE_DL_LIBS} ${X11_LIBRARIES})
      if(WIN32)
        target_link_libraries(vst2_smoke PRIVATE user32)
      else()
        target_link_options(vst2_smoke PRIVATE -Wl,--export-dynamic)
      endif()
      add_dependencies(vst2_smoke VocalBalance_VST2)
      add_test(NAME vst2_module COMMAND vst2_smoke "$<TARGET_FILE:VocalBalance_VST2>")
    endif()
  endif()
  target_link_libraries(VocalBalance PRIVATE juce::juce_audio_utils PUBLIC juce::juce_recommended_config_flags juce::juce_recommended_warning_flags)
endif()
